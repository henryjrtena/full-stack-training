Lesson 16: EF Core with API Controller

Integrate Entity Framework Core with ASP.NET Core WebAPI controllers using dependency injection and repository patterns.

Introduction

This lesson demonstrates how to combine EF Core for data access with ASP.NET Core WebAPI controllers, using dependency injection to manage the GroceryRepository. This creates a clean, testable architecture for your API endpoints.

Key Concepts

  • Repository pattern abstracts data access logic.
  • Dependency injection manages service lifetimes.
  • EF Core DbContext handles database operations.
  • API controllers use repositories for business logic.
  • Async operations ensure scalable API responses.

Repository Interface and Implementation

using GroceryApi.Models;

namespace GroceryApi.Repositories;

public interface IGroceryRepository
{
    Task<IEnumerable<GroceryItem>> GetAllAsync();
    Task<GroceryItem?> GetByIdAsync(int id);
    Task<GroceryItem> AddAsync(GroceryItem item);
    Task UpdateAsync(GroceryItem item);
    Task DeleteAsync(int id);
}

public class GroceryRepository : IGroceryRepository
{
    private readonly GroceryDbContext _context;

    public GroceryRepository(GroceryDbContext context)
    {
        _context = context;
    }

    public async Task<IEnumerable<GroceryItem>> GetAllAsync()
    {
        return await _context.GroceryItems.ToListAsync();
    }

    public async Task<GroceryItem?> GetByIdAsync(int id)
    {
        return await _context.GroceryItems.FindAsync(id);
    }

    public async Task<GroceryItem> AddAsync(GroceryItem item)
    {
        _context.GroceryItems.Add(item);
        await _context.SaveChangesAsync();
        return item;
    }

    public async Task UpdateAsync(GroceryItem item)
    {
        _context.Entry(item).State = EntityState.Modified;
        await _context.SaveChangesAsync();
    }

    public async Task DeleteAsync(int id)
    {
        var item = await _context.GroceryItems.FindAsync(id);
        if (item != null)
        {
            _context.GroceryItems.Remove(item);
            await _context.SaveChangesAsync();
        }
    }
}

Implement the repository pattern to abstract data access operations.

Program.cs with Dependency Injection

var builder = WebApplication.CreateBuilder(args);

// Add services to the container.
builder.Services.AddControllers();
builder.Services.AddDbContext<GroceryDbContext>(options =>
    options.UseSqlServer(builder.Configuration.GetConnectionString("GroceryDb")));
builder.Services.AddScoped<IGroceryRepository, GroceryRepository>();
builder.Services.AddEndpointsApiExplorer();
builder.Services.AddSwaggerGen();

var app = builder.Build();

// Configure the HTTP request pipeline.
if (app.Environment.IsDevelopment())
{
    app.UseSwagger();
    app.UseSwaggerUI();
}

app.UseHttpsRedirection();
app.UseAuthorization();
app.MapControllers();
app.Run();

Register the DbContext and repository with the DI container.

API Controller with Repository Injection

using Microsoft.AspNetCore.Mvc;
using GroceryApi.Models;
using GroceryApi.Repositories;

namespace GroceryApi.Controllers;

[ApiController]
[Route("api/[controller]")]
public class GroceriesController : ControllerBase
{
    private readonly IGroceryRepository _repository;

    public GroceriesController(IGroceryRepository repository)
    {
        _repository = repository;
    }

    [HttpGet]
    public async Task<IActionResult> GetGroceries()
    {
        var groceries = await _repository.GetAllAsync();
        return Ok(groceries);
    }

    [HttpGet("{id}")]
    public async Task<IActionResult> GetGrocery(int id)
    {
        var grocery = await _repository.GetByIdAsync(id);
        if (grocery == null)
            return NotFound();
        return Ok(grocery);
    }

    [HttpPost]
    public async Task<IActionResult> CreateGrocery([FromBody] GroceryItem grocery)
    {
        if (!ModelState.IsValid)
            return BadRequest(ModelState);

        var createdGrocery = await _repository.AddAsync(grocery);
        return CreatedAtAction(nameof(GetGrocery), new { id = createdGrocery.Id }, createdGrocery);
    }

    [HttpPut("{id}")]
    public async Task<IActionResult> UpdateGrocery(int id, [FromBody] GroceryItem grocery)
    {
        if (id != grocery.Id)
            return BadRequest();

        var existingGrocery = await _repository.GetByIdAsync(id);
        if (existingGrocery == null)
            return NotFound();

        await _repository.UpdateAsync(grocery);
        return NoContent();
    }

    [HttpDelete("{id}")]
    public async Task<IActionResult> DeleteGrocery(int id)
    {
        var grocery = await _repository.GetByIdAsync(id);
        if (grocery == null)
            return NotFound();

        await _repository.DeleteAsync(id);
        return NoContent();
    }
}

The controller uses dependency injection to access the repository for data operations.
